import type { TimelineItem } from '@granto-umbrella/umbrella-components';


type TimelineVariant =
  | 'continued'
  | 'accepted'
  | 'rejected'
  | 'issue'
  | 'order'
  | 'created'
  | 'updated'
  | 'download'
  | 'deleted'
  | 'info';


interface CompanyTimelineEvent {
  eventType: string;
  startDate: string;
  endDate: string;
  duration: number;
  isPublic: boolean;
  eventId: number;
  data: {
    eventId: string;
    entityType: string;
    entityId: number;
    eventType: string;
    eventLabel: string;
    timesTamp: string;
    actor: {
      userId: number;
      email: string;
      displayName: string;
    };
    productContext: string;
    description: string;
    changes: Array<{
      field: string;
      fieldLabel: string;
      oldValue: string;
      newValue: string;
    }>;
  };
}

// Formata a data e hora no padrão brasileiro
const formatFullTimestamp = (startDate: string, actorName: string): string => {
  const date = new Date(startDate);
  const day = String(date.getDate()).padStart(2, '0');
  const month = String(date.getMonth() + 1).padStart(2, '0');
  const year = date.getFullYear();
  const hours = String(date.getHours()).padStart(2, '0');
  const minutes = String(date.getMinutes()).padStart(2, '0');
  const seconds = String(date.getSeconds()).padStart(2, '0');

  return `Por: ${actorName} no dia ${day}/${month}/${year} às ${hours}:${minutes}:${seconds}`;
};

// Constrói o título do evento baseado no tipo
function buildTitle(ev: CompanyTimelineEvent): string {
  const eventTypeMap: Record<string, string> = {
    'UpdatedCompany': 'Empresa atualizada',
    'CreatedCompany': 'Empresa criada',
    'DeletedCompany': 'Empresa excluída',
    'Company.Updated': 'Empresa atualizada',
    'Company.Created': 'Empresa criada',
    'Company.Deleted': 'Empresa excluída',
  };

  return eventTypeMap[ev.eventType] || 
         eventTypeMap[ev.data?.eventType] || 
         ev.data?.eventLabel || 
         'Evento na empresa';
}


function getVariant(eventType: string): TimelineVariant {
  if (eventType.includes('Created') || eventType.includes('created')) {
    return 'created';
  }
  if (eventType.includes('Updated') || eventType.includes('updated')) {
    return 'updated';
  }
  if (eventType.includes('Deleted') || eventType.includes('deleted')) {
    return 'deleted';
  }
  return 'info';
}


export function mapCompanyTimelineToTimeline(items: CompanyTimelineEvent[]): TimelineItem[] {
  if (!items || !Array.isArray(items)) {
    return [];
  }

  return items.map((ev, idx) => {
    const id = ev.data?.eventId || String(idx);
    const actor = ev.data?.actor;

    const hasDetails = ev.data?.changes && ev.data.changes.length > 0;
    const comparisonData = ev.data?.changes?.map((change) => ({
      field: change.fieldLabel || change.field,
      before: change.oldValue,
      after: change.newValue,
    }));

    return {
      id,
      timestamp: formatFullTimestamp(ev.startDate, actor?.displayName || 'Sistema'),
      type: getVariant(ev.eventType),
      title: buildTitle(ev),
      actor: {
        id: String(actor?.userId || ''),
        displayName: actor?.displayName || 'Sistema',
        type: actor ? ('user' as const) : ('app' as const),
      },
      origin: ev.data?.productContext || 'Corp',
      correlationId: ev.data?.eventId || String(idx),

      metadata: {
        ...ev.data,
        eventType: ev.eventType,
        startDate: ev.startDate,
        hasDetails,
        startsExpanded: false,
        comparisonData: hasDetails ? comparisonData : undefined,
        description: ev.data?.description,
      },
    };
  });
}
