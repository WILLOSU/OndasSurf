import { useMutation } from "@tanstack/react-query";
import { api } from "../api";
import { useQuery } from "@tanstack/react-query";

interface PostOrder {
  cnpj: string;
  name: string;
  syncPiperun: boolean;
  insurers: number[];
}

export const useCreateOrder = () => {
  return useMutation({
    mutationKey: ["order"],
    mutationFn: async (order: PostOrder) => {
      const response = await api.request(
        {
          url: "/partners/orders",
          method: "POST",
          data: order,
        },
        true
      );

      return response.data;
    },
  });
};

export const useDownloadFile = () => {

      await api.download(

          method: "GET",
        },
        true
      );
    },
  });
};

interface GetOrderParams {
  insurers?: number[];
  orderId?: string | null;
  insuranceId?: string | null;
}

export const useGetOrder = () => {
  return useMutation({
    mutationFn: async ({ insurers, orderId, insuranceId }: GetOrderParams) => {
      const params: Record<string, any> = { insurers: insurers?.join(",") };

      if (orderId) params.orderId = orderId;
      if (insuranceId) params.insuranceId = insuranceId;

      const response = await api.request(
        {
          url: "/partners/orders",
          method: "GET",
          params,
        },
        true
      );

      return response.data;
    },
  });
};

interface SubscribePayload {
  insurers: number[];
  orderId: string;
  modality: string;
  alias?: string;
  data: any;
  username: string;
}

export const useSubscribe = () => {
  return useMutation({
    mutationFn: async ({
      insurers,
      orderId,
      data,
      modality,
      alias,
      username,
    }: SubscribePayload) => {
      const body = { modality, alias, form: data, username };
      const response = await api.request(
        {
          url: `/partners/subscribe/${orderId}`,
          method: "POST",
          params: {
            insurers: insurers.join(","),
          },
          data: body,
        },
        true
      );
      return response.data;
    },
  });
};

interface SubscribePayloadUpdate {

export const useUpdateSubscribe = () => {

      const response = await api.request(

            form: data,
          },
        },
        true

      return response.data;
    },
  });
};

interface TimelineParams {
  companyId?: string | number;
  proposalId?: string | number;
  continuationToken?: string;
}




export const useGetTimeline = ({ companyId, proposalId, continuationToken }: TimelineParams) => {
  return useQuery({
    queryKey: ["timeline", companyId, proposalId, continuationToken],
    queryFn: async () => {
      let url = "";
      let params = {};

      if (companyId) {
        url = `/companies/${companyId}/timeline`;
        params = { continuationToken };
      } else if (proposalId) {
        url = `/proposals/${proposalId}/timeline`;
      } else {
        throw new Error("ID necessário");
      }

      const resp = await api.request({
        url,
        method: "GET",
        params,
      });

      return resp.data;
    },
    enabled: !!companyId || !!proposalId,
    staleTime: 1000 * 60,
    refetchOnWindowFocus: false,
    refetchOnReconnect: false,
  });
};
