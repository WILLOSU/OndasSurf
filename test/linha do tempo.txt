export const Loading = () => {
  const [open, setOpen] = useState(true);
  const [loading, setLoading] = useState(true);
  const [data, setData] = useState<any[]>([]);

  useEffect(() => {
    const t = setTimeout(() => {
      setData(mapRemoteToTimeline(remoteMock));
      setLoading(false);
    }, 1500);
    return () => clearTimeout(t);
  }, []);

  return (
    <TimelineModal
      open={open}
      onOpenChange={setOpen}
      onClose={setOpen.bind(null, false)}
      data={data}
      loading={loading}
      emptyLabel="Sem eventos ainda."
    />
  );
};

export const EmptyWithRetry = () => {
  const [open, setOpen] = useState(true);
  const [loading, setLoading] = useState(false);
  const [data, setData] = useState<any[]>([]);

  const handleRetry = () => {
    setLoading(true);
    setData([]);

    setTimeout(() => {
      setData(mapRemoteToTimeline(remoteMock));
      setLoading(false);
    }, 1200);
  };

  return (
    <TimelineModal
      open={open}
      onOpenChange={setOpen}
      onClose={setOpen.bind(null, false)}
      data={data}
      loading={loading}
      onRetry={handleRetry}
      emptyLabel="Não encontramos eventos para esta proposta."
    />
  );
};

src\components\organisms\TimelineModal\TimelineModal.tsx

/* eslint-disable @typescript-eslint/no-explicit-any */
import * as Dialog from '@radix-ui/react-dialog';
import { X } from 'lucide-react';
import { CloseIcon, Content, Overlay, Title } from './TimelineModal.styles';
import { FlowArrow } from 'phosphor-react';
import { Timeline } from '../../molecules/TimeLine/TimeLine';
export const TimelineModal = ({
  open,
  onOpenChange,
  onClose,
  data,
  onRetry,
  emptyLabel,
  loading,
}: {
  open: boolean;
  onOpenChange: (open: boolean) => void;
  onClose: () => void;
  onRetry?: () => void;
  emptyLabel?: string;
  data: any;
  loading?: boolean;
}) => {
  return (
    <Dialog.Root open={open} onOpenChange={onOpenChange}>
      <Overlay />
      <Content>
        <CloseIcon aria-label="Fechar" onClick={onClose}>
          <X size={28} color="#BDBDBD" />
        </CloseIcon>

        <Dialog.Title>
          <Title>
            <FlowArrow />
            Linha do tempo
          </Title>
        </Dialog.Title>

        <Timeline
          items={data}
          onRetry={onRetry}
          emptyLabel={emptyLabel}
          skeletonItems={4}
          loading={loading}
        />
      </Content>
    </Dialog.Root>
  );
};
