import { useEffect, useState } from "react";

import { 
  TimelineModal, 
  mapRemoteToTimeline,
} from "@granto-umbrella/umbrella-components";

import type { TimelineItem } from "@granto-umbrella/umbrella-components";

import { useGetTimeline } from "@/services/endpoints/umbrella";

const TimelinePage = () => {
  const [isOpen, setIsOpen] = useState(true);
  
  
  const [timelineData, setTimelineData] = useState<TimelineItem[]>([]); 

  const { data, isLoading } = useGetTimeline({}); 

  useEffect(() => {
    if (data?.items) {
    
      setTimelineData(mapRemoteToTimeline(data.items));
    }
  }, [data]);

  const handleClose = () => {
    setIsOpen(false);
    window.history.back(); 
  };

  return (
    <TimelineModal
      open={isOpen}
      onOpenChange={setIsOpen}
      onClose={handleClose}
      data={timelineData}
      loading={isLoading}
    />
  );
};

export default TimelinePage;
